-- The following is a list of SQl functions compiled based on a ficticious healthcare dataset, showcasing additional knowledge of functions like JOINs, COUNT, GROUP BY, ORDER BY, etc. --

-- 1. Count of patients by state
SELECT State, COUNT(*) AS PatientCount
FROM patients
GROUP BY State
ORDER BY PatientCount DESC;

-- 2. Number of patients with more than one appointment
SELECT PatientName, COUNT(*) AS AppointmentCount
FROM appointments
GROUP BY PatientName
HAVING COUNT(*) > 1
ORDER BY AppointmentCount DESC;

-- 3. Patients that were prescribedd 'Metformin'
SELECT DISTINCT p.Name, p.City, p.State
FROM patients p
JOIN prescriptions pr ON p.Name = pr.PatientName
WHERE pr.Medication = 'Metformin';

-- 4. Average weight by chronic condition
SELECT ChronicConditions, AVG(WeightKG) AS AvgWeight
FROM patients
GROUP BY ChronicConditions
ORDER BY AvgWeight DESC;

-- 5. the most common department that was visited by patients 
SELECT Department, COUNT(*) AS VisitCount
FROM appointments
GROUP BY Department
ORDER BY VisitCount DESC;

-- 6. patients that do not have any perscriptions
SELECT p.Name
FROM patients p
LEFT JOIN prescriptions pr ON p.Name = pr.PatientName
WHERE pr.PatientName IS NULL;
